CREATE OR REPLACE VIEW VIEW_LAB_RESULT AS
SELECT M.LAB_RESULT_ID AS ID,
        M.HIS_LAB_NO      AS TEST_NO,
       M.REPORT_ITEM_NAME,
       M.REPORT_ITEM_CODE,
       M.ABNORMAL_INDICATOR,
       M.RESULT             AS RESULT,
       M.UNITS              AS UNITS,
       M.RESULT_DATE_TIME   AS RESULT_DATE_TIME,
       M.NORMAL_VALUE       AS REFERENCE_RESULT
  FROM MED_VIEW_LAB_RESULT M
  LEFT JOIN MED_VIEW_LAB_TEST_MASTER M1
    ON M1.TEST_ID = M.LAB_RESULT_ID;
