CREATE OR REPLACE VIEW VIEW_LAB_MASTER AS
SELECT M.TEST_NO,
       TO_CHAR(M.PRIORITY_INDICATOR) as PRIORITY_INDICATOR,
       M.VISIT_ID,
       M2.PATIENT_ID,
       M2.NAME AS NAME,
       M.TEST_CAUSE,
       M1.DIAGNOSE_ATTENTION AS RELEVANT_CLINIC_DIAG,
       M2.SOCIETY_CARD AS  SPECIMEN,
       M.RESULTS_RPT_DATE_TIME AS SPCM_RECEIVED_DATE_TIME,
       M.RESULTS_RPT_DATE_TIME AS APPLY_TIME,
       M1.DEPARTMENT_ID AS ORDERING_DEPT,
       M.ORDERING_PROVIDER AS ORDERING_PROVIDER,
       M.VERIFIED_BY AS PERFORMED_BY,
       1 RESULT_STATUS,
       M.RESULTS_RPT_DATE_TIME AS RESULTS_RPT_DATE_TIME,
        M2.SOCIETY_CARD AS  TRANSCRIPTIONIST,
       M.VERIFIED_BY VERIFIED_BY
  FROM MED_VIEW_LAB_TEST_MASTER M
  LEFT JOIN MED_CLINIC_CHECK_MASTER M1
    ON M.REG_ID = M1.REG_ID
  LEFT JOIN MED_CLINIC_CARD_INFO M2
    ON M1.INFO_ID = M2.INFO_ID;
