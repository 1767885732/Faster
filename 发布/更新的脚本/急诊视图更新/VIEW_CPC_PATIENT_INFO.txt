CREATE OR REPLACE VIEW VIEW_CPC_PATIENT_INFO AS
SELECT
'466942737' || '_' || M.REG_ID AS EHRId
,M.REGISTER_NO AS ClinicNo
,M.REG_ID AS PatientId
,C.PATIENT_ID AS EcgPatientId
,M.PHEP_PID AS PHEPPATIENTID
,M.REG_ID  AS EcgVisitId
,C.IDENTIFICATION_CARD AS IdCardNo
,'' AS CarteVitalNo
,C.NAME
,'' AS CITIZEN_CARD
,M.REG_ID AS REG_ID
,TO_CHAR(C.BIRTHDAY,'YYYY-MM-DD HH24:MI:SS') BIRTHDAY
,M.REG_AGE_CAPTION AS Age
,C.NATION
,C.GENDER
,'0' AS ISDOCARE
,m.patient_sayid AS  PATIENT_SAY
,'' AS Occupation
,'' AS Marriage
,'' AS Education
,M.GREEN_CHANNEL
,NVL(C.MOBILE_PHONE,C.LINK_MAN_TEL) AS Phone
,'' AS PHEPMISSIONID
,C.ADDRESS
,TO_CHAR(M.CREATE_TIME,'YYYY-MM-DD HH24:MI:SS') AS RegistTime
,'' AS AttackAddress
,TO_CHAR(M.REG_TIME,'YYYY-MM-DD HH24:MI:SS') AS AttackTime
,'' AS CallTime
,CASE NVL(M.PHEP_PID,'') WHEN '' THEN '0' ELSE '1' END  AS IsCallHelp
,CASE NVL(M.PHEP_PID,'') WHEN '' THEN '' ELSE '1' END AS CallSource
,'466942737' AS OrgId
, C.PATIENT_ID AS WRISTBARCODE
, M.REGISTER_NO AS VISITID
,'上海华山医院西院' AS OrgName
,'急诊工作站' AS CreateName
,TO_CHAR(SYSDATE,'YYYY-MM-DD HH24:MI:SS') AS CreateDate
FROM MED_CLINIC_CARD_INFO C
INNER JOIN MED_CLINIC_CHECK_MASTER M ON C.INFO_ID=M.INFO_ID;
