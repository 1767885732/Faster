CREATE OR REPLACE VIEW VIEW_EXAM_APPLY AS
SELECT TM.Reg_Id as PATIENT_NO,
       TM.APPLY_ID AS EXAM_NO,
       TM.EXAM_CLASS_ID AS EXAM_CLASS,
       TM.EXAM_SUBCLASS_ID AS EXAM_SUB_CLASS,
       TM.CLIN_DIAG AS CLIN_DIAG,
       TM.EXAM_MODE AS EXAM_MODE,
       TM.CREATER_ID  AS DEVICE,
       TM.CREATER_ID AS PERFORMED_BY,
       TM.CREATE_DATE AS REQ_DATE_TIME,
       TM.CREATER_ID AS REQ_DEPT,
       TM.CREATER_ID AS REQ_PHYSICIAN,
       TM.CREATER_ID AS REQ_MEMO,
       TM.EXAM_FOR AS NOTICE,
       T2.REPORT_TIME AS EXAM_START_DATE， T2.REPORT_TIME AS EXAM_END_DATE,
       T2.REPORT_TIME AS REPORT_DATE_TIME ， T1.EXAM_CHECK_NAME EXAM_PARA,
       T2.DESCRIPTION AS DESCRIPTION,
       T2.RECOMMENDATION AS RECOMMENDATION,
       TM.CREATER_ID AS TECHNICIAN,
       T2.REPORT_DOCTOR_ID REPORTER,
       TO_CHAR(TM.IS_USED) AS RESULT_STATUS,
       TM.CREATER_ID AS VERIFIED_BY,
       T2.REPORT_TIME AS VERIFIED_DATE_TIME
  FROM MED_EXAM_ITEMS_MASTER TM
  LEFT JOIN MED_EXAM_ITEMS_LIST T1
    ON TM.APPLY_ID = T1.APPLY_ID
  LEFT JOIN MED_EXAM_REPORT T2
    ON TM.APPLY_ID = T2.APPLY_ID;
